\makeatletter
\newif\if@manglingenabled
\@manglingenabledfalse

% Check if first char of #1 is uppercase (A-Z: codes 65-90)
% Usage: \@isupper{name}{\IfTrue}{\IfFalse}
\def\@isupper#1{%
    \expandafter\@isupper@check#1\@nil
}
\def\@isupper@check#1#2\@nil{%
    \ifnum`#1>64
        \ifnum`#1<91
            \expandafter\expandafter\expandafter\@firstoftwo
        \else
            \expandafter\expandafter\expandafter\@secondoftwo
        \fi
    \else
        \expandafter\@secondoftwo
    \fi
}

\def\func#1{%
    \@ifundefined{func#1}{%
        \if@manglingenabled
            \@isupper{#1}{%
                \PackageError{FunctionMapping}{No mapping found for function '#1'}{%
                    Mangling is enabled but function '#1' has no obfuscated mapping.}%
            }{%
                \texttt{#1}%
            }%
        \else
            \texttt{#1}%
        \fi
    }{%
        \allowbreak\csname func#1\endcsname\allowbreak%
    }%
}

% Include function name mapping (generated by main.js)
\IfFileExists{build/function_mapping_\myseed.tex}{%
    \input{build/function_mapping_\myseed.tex}%
    \@manglingenabledtrue
    \PackageInfo{FunctionMapping}{Loaded obfuscated function names for seed \myseed}%
}{%
    \PackageInfo{FunctionMapping}{No function mapping file - mangling disabled, using original names}%
}
\makeatother
